cmake_minimum_required(VERSION 3.16)

project(naviPlayer VERSION 0.1 LANGUAGES CXX)

set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)


option(NAVIPLAYER_BUILD_LITE "Build a simplified version" OFF)

set(FFMPEGLIB_DIR /Users/redtux/.local/libs/ffmpeg6_audio CACHE PATH "ffmpeg directory")


find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets Network Sql Multimedia Concurrent LinguistTools)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets Network Sql Multimedia Concurrent LinguistTools)
find_package(QWindowKit COMPONENTS Core Widgets REQUIRED)

set(TS_FILES naviPlayer_zh_CN.ts)

set(RC_FILES resources/naviPlayer.qrc)

if(NAVIPLAYER_BUILD_LITE)
    file(GLOB WIDGET_SRC
        src/widgets.common/*.h
        src/widgets.common/*.cpp
        src/widgets.lite/*.h
        src/widgets.lite/*.cpp
        src/widgets.lite/*.ui
    )
else()
    file(GLOB_RECURSE WIDGET_SRC
        src/widgets.common/*.h
        src/widgets.common/*.cpp
        src/widgets/*.h
        src/widgets/*.cpp
        src/widgets/*.ui
    )
endif()


file(GLOB CORE_SRC
    src/models/*.h
    src/models/*.cpp
    src/core/*.h
    src/core/*.cpp
)


set(PROJECT_SOURCES
        src/main.cpp

        ${CORE_SRC}
        ${WIDGET_SRC}
        ${RC_FILES}
        ${TS_FILES}
)


if(APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "10.15")

    #! [appicon_macOS]
    # The MACOSX_BUNDLE_ICON_FILE variable is added to the Info.plist
    # generated by CMake. This variable contains the .icns file name,
    # without the path.
    set(MACOSX_BUNDLE_ICON_FILE AppIcon.icns)

    # And the following tells CMake where to find and install the file itself.
    set(app_icon_macos "${CMAKE_CURRENT_SOURCE_DIR}/resources/icons/AppIcon.icns")

    set_source_files_properties(${app_icon_macos} PROPERTIES
        MACOSX_PACKAGE_LOCATION "Resources"
    )
    #! [appicon_macOS]
elseif (WIN32)
    #! [appicon_windows]
    set(app_icon_resource_windows "${CMAKE_CURRENT_SOURCE_DIR}/resources/naviPlayer.rc")
    #! [appicon_windows]

endif()

if(${QT_VERSION_MAJOR} GREATER_EQUAL 6)
    if (WIN32)
        qt_add_executable(naviPlayer
            MANUAL_FINALIZATION
            ${PROJECT_SOURCES}
            ${app_icon_resource_windows}
        )

    elseif (APPLE)
        qt_add_executable(naviPlayer
            MACOSX_BUNDLE
            MANUAL_FINALIZATION
            ${PROJECT_SOURCES}
            ${app_icon_macos}
        )

    else()
        qt_add_executable(naviPlayer
            MANUAL_FINALIZATION
            ${PROJECT_SOURCES}
        )

    endif()

    qt_create_translation(QM_FILES ${CMAKE_SOURCE_DIR} ${TS_FILES})

else()
    if(APPLE)
        add_executable(naviPlayer
            MACOSX_BUNDLE
            ${PROJECT_SOURCES}
            ${app_icon_macos}
        )
    elseif (WIN32)
        add_executable(naviPlayer
            ${PROJECT_SOURCES}
            ${app_icon_resource_windows}
        )

    else()
        add_executable(naviPlayer
            ${PROJECT_SOURCES}
        )
    endif()

    qt5_create_translation(QM_FILES ${CMAKE_SOURCE_DIR} ${TS_FILES})
endif()

add_subdirectory(src/qtsingleapplication)

set(FFMPEG_LIBS avcodec avformat swresample avutil)

if( CMAKE_SYSTEM_PROCESSOR MATCHES "loongarch64")
    message("this is loongarch64 cpu")
    find_path(FFMPEG_INCLUDE_PATH
        NAMES libavutil/version.h
        HINTS /usr/include/loongarch64-linux-gnu
    )
    foreach(LIB_NAME IN LISTS FFMPEG_LIBS)
        find_library(${LIB_NAME}_LIB
            NAMES ${LIB_NAME}
            HINTS /usr/lib/loongarch64-linux-gnu
        )
        list(APPEND ALL_FFMPEG_LIBS ${${LIB_NAME}_LIB})
    endforeach()

else()
    find_path(FFMPEG_INCLUDE_PATH
        NAMES libavutil/version.h
        HINTS ${FFMPEGLIB_DIR}/include
    )

    if(NOT FFMPEG_INCLUDE_PATH)
        message(FATAL_ERROR "ffmpeg lib not found. Please specify -DFFMPEGLIB_DIR=/path/to/libffmpeg")
    endif()

    foreach(LIB_NAME IN LISTS FFMPEG_LIBS)
        find_library(${LIB_NAME}_LIB
            NAMES ${LIB_NAME}
            HINTS ${FFMPEGLIB_DIR}/lib
        )
        list(APPEND ALL_FFMPEG_LIBS ${${LIB_NAME}_LIB})
    endforeach()

endif()

# 让可执行文件在运行时能找到 ffmpeg 动态库（macOS/Linux）
set(CMAKE_MACOSX_RPATH TRUE)
if(FFMPEGLIB_DIR)
    set_target_properties(naviPlayer PROPERTIES
        BUILD_RPATH "${FFMPEGLIB_DIR}/lib"
        INSTALL_RPATH "${FFMPEGLIB_DIR}/lib"
    )
endif()

target_include_directories(naviPlayer PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}/src
  ${FFMPEG_INCLUDE_PATH}
)

if(NAVIPLAYER_BUILD_LITE)
    target_include_directories(naviPlayer PRIVATE
      ${CMAKE_CURRENT_SOURCE_DIR}/src/widgets.lite
    )
else()
    target_include_directories(naviPlayer PRIVATE
      ${CMAKE_CURRENT_SOURCE_DIR}/src/widgets
      ${CMAKE_CURRENT_SOURCE_DIR}/src/widgets/pages
      ${CMAKE_CURRENT_SOURCE_DIR}/src/widgets/components
    )
endif()

target_link_libraries(naviPlayer PRIVATE
    Qt${QT_VERSION_MAJOR}::Core
    Qt${QT_VERSION_MAJOR}::Gui
    Qt${QT_VERSION_MAJOR}::Widgets
    Qt${QT_VERSION_MAJOR}::Network
    Qt${QT_VERSION_MAJOR}::Sql
    Qt${QT_VERSION_MAJOR}::Multimedia
    Qt${QT_VERSION_MAJOR}::Concurrent
    ${ALL_FFMPEG_LIBS}
)

target_link_libraries(naviPlayer PRIVATE QWindowKit::Widgets)
target_link_libraries(naviPlayer PRIVATE qtsingleapplication)

# Qt for iOS sets MACOSX_BUNDLE_GUI_IDENTIFIER automatically since Qt 6.1.
# If you are developing for iOS or macOS you should consider setting an
# explicit, fixed bundle identifier manually though.
if(${QT_VERSION} VERSION_LESS 6.1.0)
  set(BUNDLE_ID_OPTION MACOSX_BUNDLE_GUI_IDENTIFIER cn.redtux.naviPlayer)
endif()
set_target_properties(naviPlayer PROPERTIES
    ${BUNDLE_ID_OPTION}
    MACOSX_BUNDLE_BUNDLE_NAME ${PROJECT_NAME}
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}
    MACOSX_BUNDLE_INFO_STRING "subsonic music client."
    MACOSX_BUNDLE_GUI_IDENTIFIER "cn.redtux.naviPlayer"
    MACOSX_BUNDLE_COPYRIGHT "redtux@qq.com"
    MACOSX_BUNDLE TRUE
    WIN32_EXECUTABLE TRUE
)

include(GNUInstallDirs)
install(TARGETS naviPlayer
    BUNDLE DESTINATION .
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)

if(QT_VERSION_MAJOR EQUAL 6)
    qt_finalize_executable(naviPlayer)
endif()
